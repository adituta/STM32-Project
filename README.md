/* RCC APB1 - pentru USART2 */
#define RCC_APB1ENR  (*(volatile uint32_t *)0x40023840)

/* GPIOA Alternate Function Low register (pini 0-7) */
#define GPIOA_AFRL   (*(volatile uint32_t *)0x40020020)

/* USART2 */
#define USART2_SR   (*(volatile uint32_t *)0x40004400)
#define USART2_DR   (*(volatile uint32_t *)0x40004404)
#define USART2_BRR  (*(volatile uint32_t *)0x40004408)
#define USART2_CR1  (*(volatile uint32_t *)0x4000440C)

#define USART_SR_TXE  (1 << 7)
#define USART_CR1_UE  (1 << 13)
#define USART_CR1_TE  (1 << 3)

static void uart2_init(void)
{
    /* activeaza ceasul pentru USART2 (bus APB1, bit 17) */
    RCC_APB1ENR |= (1 << 17);

    /* PA2 = TX, PA3 = RX -> mod Alternate Function (10) */
    GPIOA_MODER &= ~((0x3 << (2*2)) | (0x3 << (3*2)));
    GPIOA_MODER |=  ((0x2 << (2*2)) | (0x2 << (3*2)));

    /* AFRL: 4 biti per pin. PA2 -> biti 8-11, PA3 -> biti 12-15.
       AF7 = 0111 pentru USART2 */
    GPIOA_AFRL &= ~((0xF << (2*4)) | (0xF << (3*4)));
    GPIOA_AFRL |=  ((0x7 << (2*4)) | (0x7 << (3*4)));

    /* baud rate: 16MHz / 115200 ~= 139 */
    USART2_BRR = 139;

    /* activeaza USART + transmisie */
    USART2_CR1 = USART_CR1_UE | USART_CR1_TE;
}

static void uart2_send_char(char c)
{
    /* asteapta pana registrul de date e liber (TXE=1) */
    while (!(USART2_SR & USART_SR_TXE));
    USART2_DR = c;
}

static void uart2_send_string(const char *s)
{
    while (*s)
    {
        uart2_send_char(*s);
        s++;
    }
}

static void uint32_to_string(uint32_t nr, unsigned char* out){
	//zicem ca stringul are o dimensiune suficienta ca sa incapa datele pe el
	// trebuie sa extrag cifrele si sa le pun in string
	unsigned char buffer[20];

	int  i =0;
	while(nr!=0){
		buffer[i] = nr%10;
		i++;
		nr = nr/10;
	}

	// acum trebuie sa inversez numerele in buffer
	for(int j = 0;j<i/2;j++){
		//trebuie sa inversez ordinea lor
		unsigned char c = buffer[j];
		buffer[j] = buffer[i-1-j];
		buffer[i-1-j] = c;
	}

	//acum trebuie sa convertesc totul la ascii
	//trebuie sa am grija
	for(int j = 0;j<i;j++)
		buffer[j]+= '0';

	//facem un memcpy in bufferul de afisare
	//pun si caracterele null si endline
	buffer[i] = '\r';
	i++;
	buffer[i] = '\n';
	i++;
	buffer[i] = '\0';
	i++;
	memcpy(out, buffer, i);


}
